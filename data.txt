/**
 * Problem   : Two Sum
 * Source    : LeetCode #1
 * Difficulty: Easy
 * Link      : https://leetcode.com/problems/two-sum/
 *
 * Description:
 *   Given an array of integers nums and an integer target,
 *   return indices of the two numbers that add up to target.
 *
 * Constraints:
 *   - 2 <= nums.length <= 10^4
 *   - Each input has exactly one solution
 *   - You may not use the same element twice
 */
public class TwoSum {

    // ── Approach 1: Brute Force ──────────────────────────────
    // Time: O(n²) | Space: O(1)
    // Check every pair — correct but too slow for large inputs
    public static int[] solveBrute(int[] nums, int target) {
        for (int i = 0; i < nums.length; i++) {
            for (int j = i + 1; j < nums.length; j++) {
                if (nums[i] + nums[j] == target) {
                    return new int[]{i, j};
                }
            }
        }
        return new int[]{};
    }

    // ── Approach 2: HashMap (Optimal) ────────────────────────
    // Time: O(n) | Space: O(n)
    // Store each number's index; look up complement on each step
    public static int[] solveOptimal(int[] nums, int target) {
        java.util.HashMap<Integer, Integer> map = new java.util.HashMap<>();
        for (int i = 0; i < nums.length; i++) {
            int complement = target - nums[i];
            if (map.containsKey(complement)) {
                return new int[]{map.get(complement), i};
            }
            map.put(nums[i], i);
        }
        return new int[]{};
    }

    // ── Main: test both approaches ───────────────────────────
    public static void main(String[] args) {
        // Test 1: normal case
        int[] r1 = solveOptimal(new int[]{2, 7, 11, 15}, 9);
        System.out.println("Test 1: [" + r1[0] + ", " + r1[1] + "]"); // [0, 1]

        // Test 2: answer is not the first two elements
        int[] r2 = solveOptimal(new int[]{3, 2, 4}, 6);
        System.out.println("Test 2: [" + r2[0] + ", " + r2[1] + "]"); // [1, 2]

        // Test 3: same value appears at two different indices
        int[] r3 = solveOptimal(new int[]{3, 3}, 6);
        System.out.println("Test 3: [" + r3[0] + ", " + r3[1] + "]"); // [0, 1]
    }
}`

function Checklist({ items }) {
  const [done, setDone] = useState({});
  const count = Object.values(done).filter(Boolean).length;
  return (
    <div>
      <div style={{ display: "flex", flexDirection: "column", gap: "6px" }}>
        {items.map((item, i) => (
          <div key={i} onClick={() => setDone(p => ({ ...p, [i]: !p[i] }))} style={{
            background: done[i] ? "#0a1e12" : "#090e1a",
            border: `1px solid ${done[i] ? "#1a5c3050" : "#151f30"}`,
            borderRadius: "10px", padding: "11px 14px", cursor: "pointer",
            display: "flex", gap: "12px", alignItems: "flex-start", transition: "all 0.15s"
          }}>
            <div style={{
              width: "17px", height: "17px", borderRadius: "4px", flexShrink: 0, marginTop: "2px",
              background: done[i] ? "#15803d" : "transparent",
              border: `2px solid ${done[i] ? "#15803d" : "#2a3a50"}`,
              display: "flex", alignItems: "center", justifyContent: "center"
            }}>
              {done[i] && <span style={{ fontSize: "10px", color: "#fff", fontWeight: 700 }}>✓</span>}
            </div>
            <div>
              <div style={{ fontWeight: 600, fontSize: "13px", color: done[i] ? "#6ee7a0" : "#bfccd8", marginBottom: "3px", textDecoration: done[i] ? "line-through" : "none", opacity: done[i] ? 0.65 : 1 }}>{item.step}</div>
              <div style={{ fontSize: "12px", color: "#4a6070", lineHeight: "1.7" }}>{item.detail}</div>
            </div>
          </div>
        ))}
      </div>
      <div style={{ fontSize: "11.5px", color: "#2a3a50", textAlign: "right", marginTop: "7px" }}>
        {count}/{items.length} done {count === items.length ? "🎉" : ""}
      </div>
    </div>
  );
}
